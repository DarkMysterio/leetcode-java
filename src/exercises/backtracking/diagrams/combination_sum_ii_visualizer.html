public List<List<Integer>> combinationSum2(int[] candidates, int target) {

void backtrack(res, path, target, sum, startIndex, candidates) {
if (currentSum > target) return; // Prune exceed
if (currentSum == target) { result.add(new ArrayList<>(path)); return; }
for (int i = startIndex; i < candidates.length; i++) {
// DUPLICATE SKIP RULE
if (i > startIndex && candidates[i] == candidates[i - 1]) continue;
if (currentSum + candidates[i] > target) break; // Early loop break
path.add(candidates[i]); // 1. CHOOSE
backtrack(..., sum + candidates[i], i + 1); // 2. RECURSE (i + 1)
path.remove(path.size() - 1); // 3. UN-CHOOSE (Backtrack)
}
}
