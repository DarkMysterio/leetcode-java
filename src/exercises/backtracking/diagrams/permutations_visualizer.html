Backtracking Pattern
public List<List<Integer>> permute(int[] nums) {
backtrack(result, new ArrayList<>(), nums);
}
void backtrack(result, currentPath, nums) {
if (currentPath.size() == nums.length) {
result.add(new ArrayList<>(currentPath)); return;
}
// ALWAYS LOOP FROM 0 FOR PERMUTATIONS!
for (int i = 0; i < nums.length; i++) {
if (!currentPath.contains(nums[i])) {
currentPath.add(nums[i]); // 1. CHOOSE
backtrack(result, currentPath, nums); // 2. RECURSE
currentPath.remove(currentPath.size() - 1); // 3. UN-CHOOSE
}
}
}
