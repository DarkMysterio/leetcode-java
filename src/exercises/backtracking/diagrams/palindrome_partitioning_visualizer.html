public void backtrack(result, currentArray, index, s) {
if (index == s.length()) { // Base Case
if (allPalindromes(currentArray)) result.add(new ArrayList<>(currentArray));
return;
}
// CHOICE 1: CUT (New Substring)
currentArray.add(String.valueOf(s.charAt(index)));
backtrack(result, currentArray, index + 1, s);
currentArray.remove(currentArray.size() - 1); // Backtrack
// CHOICE 2: MERGE (Append to Last Substring)
int lastIdx = currentArray.size() - 1;
String orig = currentArray.get(lastIdx);
currentArray.set(lastIdx, orig + s.charAt(index));
backtrack(result, currentArray, index + 1, s);
currentArray.set(lastIdx, orig); // Backtrack String
}
