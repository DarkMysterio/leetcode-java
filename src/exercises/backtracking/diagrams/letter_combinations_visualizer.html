public void backtrack(result, currentPath, phoneDigits, index, digits) {
if (index == digits.length()) { // Base Case
result.add(currentPath.toString());
return;
}
int currentDigit = digits.charAt(index) - '0';
String letters = phoneDigits.get(currentDigit);
for (int i = 0; i < letters.length(); i++) {
currentPath.append(letters.charAt(i)); // 1. CHOOSE
backtrack(result, currentPath, phoneDigits, index + 1, digits); // 2. RECURSE
currentPath.deleteCharAt(currentPath.length() - 1); // 3. UN-CHOOSE
}
}
