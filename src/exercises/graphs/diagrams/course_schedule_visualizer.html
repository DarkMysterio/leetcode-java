public boolean canFinish(int numCourses, int[][] prerequisites) {
// 1. Pre-populate map for ALL courses 0 to numCourses - 1
HashMap<Integer, List<Integer>> courses = new HashMap<>();
for (int i = 0; i < numCourses; i++) courses.put(i, new ArrayList<>());
// 2. Build adjacency list
for (int[] pre : prerequisites) courses.get(pre[0]).add(pre[1]);
// 3. Track active DFS stack
HashSet<Integer> visited = new HashSet<>();
// 4. Run DFS on every course
for (int c = 0; c < numCourses; c++) {
if (!dfs(c, visited, courses)) return false;
} return true;
}
boolean dfs(int currentCourse, HashSet<Integer> visited, Map...) {
if (visited.contains(currentCourse)) return false; // CYCLE DETECTED!
List<Integer> preMap = courses.get(currentCourse);
if (preMap.isEmpty()) return true; // MEMOIZED SAFE / NO PREREQS
visited.add(currentCourse); // Add to active stack
for (Integer pre : preMap) {
if (!dfs(pre, visited, courses)) return false;
}
visited.remove(currentCourse); // Backtrack active stack
courses.put(currentCourse, new ArrayList<>()); // MEMOIZE SAFE
return true;
}
