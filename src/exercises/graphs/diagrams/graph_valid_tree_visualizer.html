public boolean validTree(int n, int[][] edges) {
// 1. O(1) Edge Count Theorem: Trees MUST have N - 1 edges
if (edges.length != n - 1) return false;
// 2. Build Undirected Adjacency List
List<List<Integer>> adjList = new ArrayList<>();
for (int i = 0; i < n; i++) adjList.add(new ArrayList<>());
for (int[] edge : edges) { adjList.get(edge[0]).add(edge[1]); ... }
HashSet<Integer> visited = new HashSet<>();
// 3. Start DFS from node 0 to measure reachability
dfs(0, adjList, visited);
// 4. Graph is a tree IF AND ONLY IF all N nodes are visited!
return visited.size() == n;
}
void dfs(int node, List<List<Integer>> adjList, HashSet<Integer> visited) {
if (visited.contains(node)) return; // Avoid infinite loops!
visited.add(node); // Mark node visited
for (Integer neighbor : adjList.get(node)) {
dfs(neighbor, adjList, visited);
}
}
