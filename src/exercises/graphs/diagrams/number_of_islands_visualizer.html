public int numIslands(char[][] grid) {
boolean[][] visited = new boolean[rows][cols];
for (int i = 0; i < rows; i++) {
for (int j = 0; j < cols; j++) {
if (grid[i][j] == '1' && !visited[i][j]) {
count++; dfs(grid, visited, i, j);
}
}
} return count;
}
void dfs(grid, visited, i, j) {
if (outOfBounds || grid[i][j] == '0' || visited[i][j]) return;
visited[i][j] = true; // Mark visited
dfs(i + 1, j); dfs(i - 1, j); // Down, Up
dfs(i, j + 1); dfs(i, j - 1); // Right, Left
}
