public boolean checkInclusion(String s1, String s2) {
if (s1 == null || s2 == null || s1.length() > s2.length()) return false;
// 1. Construim s1_count map pentru s1
HashMap<Character, Integer> s1_count = new HashMap<>(26);
for (char c : s1.toCharArray()) s1_count.put(c, s1_count.getOrDefault(c, 0) + 1);
// 2. Inițializăm window_count pentru prima fereastră [0...wSize-1]
HashMap<Character, Integer> window_count = new HashMap<>(26);
for (int i = 0; i < wSize; i++) window_count.put(s2.charAt(i), ...);
if (window_count.equals(s1_count)) return true;
// 3. Glisăm fereastra de la indexul 'wSize' până la s2.length() - 1
for (int r = wSize; r < s2.length(); r++) {
// ADAUGĂM caracterul care intră în dreapta
char charIn = s2.charAt(r);
window_count.put(charIn, window_count.getOrDefault(charIn, 0) + 1);
// SCOATEM caracterul care iese prin stânga
char charOut = s2.charAt(r - wSize);
window_count.put(charOut, window_count.get(charOut) - 1);
// CURĂȚARE CHEIE CU VECIN 0!
if (window_count.get(charOut) == 0) window_count.remove(charOut);
// Verificăm dacă Map-urile sunt egale
if (window_count.equals(s1_count)) return true;
}
return false;
}
